# demo fictional config - FortiGate FortiOS
#config-version=FGT60E-5.06-FW-build1634-210301
config system global
    set hostname "FG-BRANCH-01"
    set admintimeout 60
    set admin-sport 443
    set admin-https-redirect enable
end
config system interface
    edit "port1"
        set ip 198.51.100.10 255.255.255.248
        set allowaccess ping https ssh
        set description "WAN ISP-A"
    next
    edit "port2"
        set ip 192.168.40.1 255.255.255.0
        set allowaccess ping https http ssh snmp
        set description "Branch LAN"
    next
    edit "port3"
        set ip 10.60.0.1 255.255.0.0
        set allowaccess ping
        set description "Lab network"
    next
end
config router static
    edit 1
        set dst 0.0.0.0 0.0.0.0
        set gateway 198.51.100.9
        set device port1
    next
end
config system admin
    edit "admin"
        set trusthost1 0.0.0.0 0.0.0.0
        set accprofile "super_admin"
    next
end
config system snmp community
    edit 1
        set name "public"
        set status enable
    next
end
config firewall policy
    edit 1
        set name "WAN to LAN full"
        set srcintf "port1"
        set dstintf "port2"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set logtraffic disable
    next
    edit 2
        set name "LAN to WAN"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
        set logtraffic utm
    next
end
